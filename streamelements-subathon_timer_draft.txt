StreamElements Custom Widget – Teewee Long Stream Timer (Draft)

This widget is designed as a bottom-centre timer rail for a long stream / subathon-style session.
It matches the existing teewee overlay language:
- Dark transparent broadcast overlay
- Deep green / emerald brand palette
- Block Script for the hero timer
- Clean lower-third layout with rules panel
- Event-coloured add-time feedback

Behaviour locked in from our discussion:
- Starts at 6 hours
- Counts down continuously
- Shows negative time after zero (overtime)
- Follow = +10 minutes
- Subscribed = +20 minutes
- Prime sub = +20 minutes
- Gifted subs = +20 minutes per gifted sub, counting the bulk gifter event only
- 100 bits = +5 minutes per full 100 bits
- Uses localStorage persistence so a refresh does not wipe the timer

Paste each block into the matching StreamElements Custom Widget tab:
- Fields
- HTML
- CSS
- JS
- Data

const STORAGE_KEY = 'teewee_long_stream_timer_v1';

const state = {
  fieldData: {},
  debugMode: false,
  startingHours: 6,
  followMinutes: 10,
  subMinutes: 20,
  giftSubMinutes: 20,
  bitsStep: 100,
  bitsStepMinutes: 5,
  startedDurationMs: 6 * 60 * 60 * 1000,
  targetAt: 0,
  popTimer: 0,
  ruleTimers: new Map(),
  fingerprints: new Map(),
  lastBulkGift: {
    at: 0,
    sender: '',
    amount: 0
  }
};

const els = {};

function $(id) {
  return document.getElementById(id);
}

function cacheEls() {
  els.root = $('timer-root');
  els.timerDisplay = $('timer-display');
  els.timerSign = $('timer-sign');
  els.eventPop = $('event-pop');
  els.adminControls = $('admin-controls');
  els.ruleFollow = $('rule-follow');
  els.ruleSub = $('rule-sub');
  els.ruleGift = $('rule-gift');
  els.ruleBits = $('rule-bits');
  els.ruleFollowValue = $('rule-follow-value');
  els.ruleSubValue = $('rule-sub-value');
  els.ruleGiftValue = $('rule-gift-value');
  els.ruleBitsValue = $('rule-bits-value');
}

function logDebug(label, value) {
  if (!state.debugMode) return;
  console.log('[Teewee Timer]', label, value);
}

function setCssVar(name, value) {
  if (!value) return;
  document.documentElement.style.setProperty(name, value);
}

function formatClock(ms) {
  const totalSeconds = Math.floor(Math.abs(ms) / 1000);
  const hours = Math.floor(totalSeconds / 3600);
  const minutes = Math.floor((totalSeconds % 3600) / 60);
  const seconds = totalSeconds % 60;
  return [hours, minutes, seconds].map((v) => String(v).padStart(2, '0')).join(':');
}

function minutesToMs(minutes) {
  return Math.max(0, Number(minutes) || 0) * 60 * 1000;
}

function saveState() {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify({
      targetAt: state.targetAt,
      startedDurationMs: state.startedDurationMs
    }));
  } catch (error) {
    logDebug('saveState failed', error);
  }
}

function loadState() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return false;
    const parsed = JSON.parse(raw);
    if (!parsed || typeof parsed !== 'object') return false;
    if (!Number.isFinite(parsed.targetAt)) return false;
    state.targetAt = parsed.targetAt;
    if (Number.isFinite(parsed.startedDurationMs)) {
      state.startedDurationMs = parsed.startedDurationMs;
    }
    return true;
  } catch (error) {
    logDebug('loadState failed', error);
    return false;
  }
}

function resetTimer() {
  state.startedDurationMs = state.startingHours * 60 * 60 * 1000;
  state.targetAt = Date.now() + state.startedDurationMs;
  saveState();
  renderTimer();
}

function ensureTimer() {
  if (loadState()) {
    renderTimer();
    return;
  }
  resetTimer();
}

function renderTimer() {
  const remainingMs = state.targetAt - Date.now();
  els.timerSign.textContent = remainingMs < 0 ? '-' : '';
  els.timerDisplay.textContent = formatClock(remainingMs);
}

function tick() {
  renderTimer();
  window.requestAnimationFrame(() => {
    window.setTimeout(tick, 250);
  });
}

function cleanupFingerprints() {
  const now = Date.now();
  for (const [key, timestamp] of state.fingerprints.entries()) {
    if (now - timestamp > 8000) {
      state.fingerprints.delete(key);
    }
  }
}

function buildFingerprint(listener, payload) {
  const data = payload && payload.data ? payload.data : {};
  const event = payload && payload.event ? payload.event : {};

  return [
    listener,
    data._id,
    data.id,
    data.name,
    data.username,
    data.sender,
    data.gifted,
    data.amount,
    data.createdAt,
    event._id,
    event.id
  ].filter(Boolean).join('|');
}

function isDuplicateEvent(listener, payload) {
  cleanupFingerprints();

  const fingerprint = buildFingerprint(listener, payload);
  if (!fingerprint) return false;

  const now = Date.now();
  const previous = state.fingerprints.get(fingerprint);
  state.fingerprints.set(fingerprint, now);

  return Boolean(previous && now - previous < 3000);
}

function normaliseDetail(detail) {
  const event = detail && typeof detail.event === 'object' ? detail.event : {};
  const eventData = event && typeof event.data === 'object' ? event.data : {};

  return {
    listener: detail && detail.listener ? detail.listener : '',
    type: detail && detail.type ? detail.type : '',
    event,
    data: {
      ...(detail || {}),
      ...(event || {}),
      ...(eventData || {})
    }
  };
}

function safeText(value) {
  return typeof value === 'string' ? value.trim() : '';
}

function resolveUsername(payload, fallback = 'Someone') {
  const data = payload && payload.data ? payload.data : {};
  const event = payload && payload.event ? payload.event : {};
  const candidates = [
    data.name,
    data.username,
    data.displayName,
    data.display_name,
    data.userName,
    data.senderName,
    event.name,
    event.username,
    event.displayName,
    event.display_name
  ];

  for (const candidate of candidates) {
    const text = safeText(candidate);
    if (text) return text;
  }

  return fallback;
}

function resolveNumber(payload, keys, fallback = 0) {
  const data = payload && payload.data ? payload.data : {};
  const event = payload && payload.event ? payload.event : {};

  for (const key of keys) {
    const value = data[key] ?? event[key];
    const parsed = Number(value);
    if (Number.isFinite(parsed)) return parsed;
  }

  return fallback;
}

function looksPrime(payload) {
  const data = payload && payload.data ? payload.data : {};
  const event = payload && payload.event ? payload.event : {};
  const candidates = [
    data.isPrime,
    data.prime,
    data.plan,
    data.planName,
    data.subPlan,
    data.subscriptionPlan,
    data.tier,
    data.message,
    event.isPrime,
    event.prime,
    event.plan,
    event.planName,
    event.subPlan,
    event.subscriptionPlan,
    event.tier,
    event.message
  ];

  return candidates.some((value) => {
    if (value === true) return true;
    if (typeof value !== 'string' && typeof value !== 'number') return false;
    return String(value).toLowerCase().includes('prime');
  });
}

function addTime(minutes, meta) {
  const addMs = minutesToMs(minutes);
  if (!addMs) return;
  state.targetAt += addMs;
  saveState();
  renderTimer();
  showEventPop(meta.type, meta.label);
  pulseRule(meta.ruleEl);
}

function showEventPop(type, text) {
  window.clearTimeout(state.popTimer);
  els.eventPop.classList.remove('is-visible');
  void els.eventPop.offsetWidth;
  els.eventPop.dataset.type = type;
  els.eventPop.textContent = text;
  els.eventPop.classList.add('is-visible');
  state.popTimer = window.setTimeout(() => {
    els.eventPop.classList.remove('is-visible');
  }, 2400);
}

function pulseRule(el) {
  if (!el) return;
  const previous = state.ruleTimers.get(el);
  if (previous) window.clearTimeout(previous);
  el.classList.remove('is-hit');
  void el.offsetWidth;
  el.classList.add('is-hit');
  state.ruleTimers.set(el, window.setTimeout(() => {
    el.classList.remove('is-hit');
    state.ruleTimers.delete(el);
  }, 900));
}

function updateRuleLabels() {
  els.ruleFollowValue.textContent = `+${state.followMinutes}m`;
  els.ruleSubValue.textContent = `+${state.subMinutes}m`;
  els.ruleGiftValue.textContent = `+${state.giftSubMinutes}m each`;
  els.ruleBitsValue.textContent = `+${state.bitsStepMinutes}m`;
  const ruleBitsName = els.ruleBits ? els.ruleBits.querySelector('.rule-name') : null;
  if (ruleBitsName) ruleBitsName.textContent = `${state.bitsStep} Bits`;
}

function applyFieldData(fieldData) {
  state.fieldData = fieldData || {};
  state.debugMode = Boolean(fieldData.debugMode);
  state.startingHours = Number(fieldData.startingHours) || 6;
  state.followMinutes = Number(fieldData.followMinutes) || 10;
  state.subMinutes = Number(fieldData.subMinutes) || 20;
  state.giftSubMinutes = Number(fieldData.giftSubMinutes) || 20;
  state.bitsStep = Number(fieldData.bitsStep) || 100;
  state.bitsStepMinutes = Number(fieldData.bitsStepMinutes) || 5;

  setCssVar('--follow-colour', fieldData.followColour || '#053F33');
  setCssVar('--subscribed-colour', fieldData.subscribedColour || '#04473C');
  setCssVar('--prime-sub-colour', fieldData.primeSubColour || '#00A8E1');
  setCssVar('--gifted-subs-colour', fieldData.giftedSubsColour || '#2ED9B0');
  setCssVar('--bits-colour', fieldData.bitsColour || '#F5B301');

  els.adminControls.hidden = !Boolean(fieldData.showAdmin);
  updateRuleLabels();
}

function handleFollow() {
  addTime(state.followMinutes, {
    type: 'follow',
    label: `FOLLOW +${state.followMinutes} MIN`,
    ruleEl: els.ruleFollow
  });
}

function handleBits(payload) {
  const amount = Math.floor(resolveNumber(payload, ['amount', 'count'], 0));
  const chunks = Math.floor(amount / state.bitsStep);
  if (chunks <= 0) return;
  const minutes = chunks * state.bitsStepMinutes;
  addTime(minutes, {
    type: 'bits',
    label: `${amount} BITS +${minutes} MIN`,
    ruleEl: els.ruleBits
  });
}

function handleSub(payload) {
  const username = resolveUsername(payload);
  const type = looksPrime(payload) ? 'prime-sub' : 'subscribed';
  const labelPrefix = looksPrime(payload) ? 'PRIME' : 'SUB';
  addTime(state.subMinutes, {
    type,
    label: `${labelPrefix} +${state.subMinutes} MIN`,
    ruleEl: els.ruleSub
  });
  logDebug('Handled sub', { username, type });
}

function handleGiftBulk(payload) {
  const gifter = resolveUsername(payload);
  const amount = Math.max(1, Math.floor(resolveNumber(payload, ['amount', 'count'], 1)));
  const minutes = amount * state.giftSubMinutes;

  state.lastBulkGift = {
    at: Date.now(),
    sender: gifter.toLowerCase(),
    amount
  };

  addTime(minutes, {
    type: 'gifted-subs',
    label: `${gifter} +${minutes} MIN`,
    ruleEl: els.ruleGift
  });
}

function isGiftRecipientEcho(payload) {
  const sender = safeText((payload.data && payload.data.sender) || (payload.event && payload.event.sender)).toLowerCase();
  if (!sender) return false;
  const recent = state.lastBulkGift;
  return Boolean(recent.sender && sender === recent.sender && Date.now() - recent.at < 8000);
}

function handleIncoming(detail) {
  const payload = normaliseDetail(detail);
  const listener = payload.listener;

  window.TeeweeTimerLastEvent = detail;
  logDebug('Incoming detail', detail);
  logDebug('Normalised payload', payload);

  if (!listener) return;

  const supported = [
    'follower-latest',
    'cheer-latest',
    'subscriber-latest',
    'subscriber-new-latest',
    'subscriber-resub-latest',
    'subscriber-gifted-latest'
  ];

  if (!supported.includes(listener)) return;
  if (isDuplicateEvent(listener, payload)) return;

  if (listener === 'follower-latest') {
    handleFollow();
    return;
  }

  if (listener === 'cheer-latest') {
    handleBits(payload);
    return;
  }

  if (listener === 'subscriber-gifted-latest') {
    handleGiftBulk(payload);
    return;
  }

  if (
    listener === 'subscriber-latest' ||
    listener === 'subscriber-new-latest' ||
    listener === 'subscriber-resub-latest'
  ) {
    const data = payload.data || {};
    const isGiftDerived = Boolean(
      data.sender ||
      data.gifted ||
      data.isGift ||
      data.isGifted ||
      data.bulkGifted
    );

    if (isGiftDerived && isGiftRecipientEcho(payload)) {
      logDebug('Skipped gift recipient echo', payload);
      return;
    }

    if (isGiftDerived) {
      return;
    }

    handleSub(payload);
  }
}

function bindAdminControls() {
  if (!els.adminControls) return;
  els.adminControls.addEventListener('click', (event) => {
    const target = event.target;
    if (!(target instanceof HTMLElement)) return;
    const action = target.getAttribute('data-action');
    if (!action) return;

    if (action === 'plus-follow') {
      handleFollow();
      return;
    }
    if (action === 'plus-sub') {
      addTime(state.subMinutes, {
        type: 'subscribed',
        label: `SUB +${state.subMinutes} MIN`,
        ruleEl: els.ruleSub
      });
      return;
    }
    if (action === 'plus-gift') {
      addTime(state.giftSubMinutes, {
        type: 'gifted-subs',
        label: `GIFT +${state.giftSubMinutes} MIN`,
        ruleEl: els.ruleGift
      });
      return;
    }
    if (action === 'plus-bits') {
      addTime(state.bitsStepMinutes, {
        type: 'bits',
        label: `${state.bitsStep} BITS +${state.bitsStepMinutes} MIN`,
        ruleEl: els.ruleBits
      });
      return;
    }
    if (action === 'reset') {
      resetTimer();
    }
  });
}

window.addEventListener('onWidgetLoad', function (obj) {
  cacheEls();
  applyFieldData((obj.detail && obj.detail.fieldData) || {});
  bindAdminControls();
  ensureTimer();
  tick();

  window.TeeweeTimerReset = resetTimer;
  window.TeeweeTimerAddMinutes = function (minutes) {
    addTime(Number(minutes) || 0, {
      type: 'follow',
      label: `+${Number(minutes) || 0} MIN`,
      ruleEl: null
    });
  };
  window.TeeweeTimerState = state;
});

window.addEventListener('onEventReceived', function (obj) {
  if (!obj || !obj.detail) return;
  handleIncoming(obj.detail);
});

window.addEventListener('onSessionUpdate', function (obj) {
  if (!obj || !obj.detail) return;
  handleIncoming(obj.detail);
});
