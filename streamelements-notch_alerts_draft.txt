StreamElements Custom Widget – Teewee Notch Alerts (Draft)

Replacement for the existing centre-screen dramatic alerts. Designed to sit in
the same exact top-centred position as the "fully down" notch overlay, using
the same physical dimensions (520 x 74 px) and the same ease, but without
the notch pill chrome.

Differences vs. the existing alltypes draft:
- Same total hold duration by default (6200 ms) so alerts feel identical in
  terms of how long a name stays on screen.
- No pill, no background, no rays — just the icon + Block Script headline.
- Correct SVG per alert type, pulled from teewee.live/screens/assets.
- One shared sound for every event type: the follow.wav sound.
- One dedupe/queue engine per event family so gifted sub spam matches the
  recipient-style alert contract you settled on earlier.

Paste each block into the matching StreamElements Custom Widget tab:
- Fields
- HTML
- CSS
- JS
- Data

function logDebug(label, value) {
  if (!state.debugMode) return;
  console.log(`[Teewee Notch Alerts] ${label}`, value);
}

function setCssVar(name, value) {
  if (value === undefined || value === null || value === '') return;
  document.documentElement.style.setProperty(name, value);
}

function rgbToRgbaString(hex, alpha) {
  const clean = String(hex || '').replace('#', '').trim();
  if (clean.length !== 3 && clean.length !== 6) return '';
  const full = clean.length === 3
    ? clean.split('').map((c) => c + c).join('')
    : clean;
  const r = parseInt(full.slice(0, 2), 16);
  const g = parseInt(full.slice(2, 4), 16);
  const b = parseInt(full.slice(4, 6), 16);
  if (![r, g, b].every(Number.isFinite)) return '';
  return `rgba(${r}, ${g}, ${b}, ${alpha})`;
}

function applyFieldData(fieldData) {
  if (!fieldData) return;

  state.duration = Number(fieldData.displayDuration) || 6200;
  state.debugMode = Boolean(fieldData.debugMode);
  state.soundVolume = Math.min(1, Math.max(0, Number(fieldData.soundVolume) / 100 || 0));
  state.soundUrl = (fieldData.soundUrl || '').trim() || 'https://teewee.live/sounds/follow.wav';
  state.notchOffsetTop = Number.isFinite(Number(fieldData.iconOffsetTopPx))
    ? Number(fieldData.iconOffsetTopPx)
    : 18;

  state.iconUrls.follow = (fieldData.iconFollow || '').trim();
  state.iconUrls.bits = (fieldData.iconBits || '').trim();
  state.iconUrls.subscribed = (fieldData.iconSubscribed || '').trim();
  state.iconUrls['prime-sub'] = (fieldData.iconPrimeSub || '').trim();
  state.iconUrls['gifted-subs'] = (fieldData.iconGiftedSubs || '').trim();
  state.iconUrls.raid = (fieldData.iconRaid || '').trim();
  state.iconUrls.hypetrain = (fieldData.iconHypeTrain || '').trim();

  setCssVar('--alert-duration', `${state.duration}ms`);
  setCssVar('--notch-offset-top', `${state.notchOffsetTop}px`);

  if (fieldData.accentFollow) {
    setCssVar('--accent-follow', String(fieldData.accentFollow));
    setCssVar('--glow-follow', rgbToRgbaString(fieldData.accentFollow, 0.32));
  }
  if (fieldData.accentBits) {
    setCssVar('--accent-bits', String(fieldData.accentBits));
    setCssVar('--glow-bits', rgbToRgbaString(fieldData.accentBits, 0.32));
  }
  if (fieldData.accentSubscribed) {
    setCssVar('--accent-subscribed', String(fieldData.accentSubscribed));
    setCssVar('--glow-subscribed', rgbToRgbaString(fieldData.accentSubscribed, 0.32));
  }
  if (fieldData.accentPrimeSub) {
    setCssVar('--accent-prime-sub', String(fieldData.accentPrimeSub));
    setCssVar('--glow-prime-sub', rgbToRgbaString(fieldData.accentPrimeSub, 0.32));
  }
  if (fieldData.accentGiftedSubs) {
    setCssVar('--accent-gifted-subs', String(fieldData.accentGiftedSubs));
    setCssVar('--glow-gifted-subs', rgbToRgbaString(fieldData.accentGiftedSubs, 0.34));
  }
  if (fieldData.accentRaid) {
    setCssVar('--accent-raid', String(fieldData.accentRaid));
    setCssVar('--glow-raid', rgbToRgbaString(fieldData.accentRaid, 0.28));

    els.testPanel.hidden = !state.debugMode;
  }
}

function playSoundOnce(url) {
  if (!url) return;
  try {
    const audio = new Audio(url);
    audio.volume = state.soundVolume;
    audio.preload = 'auto';
    const playAttempt = audio.play();
    if (playAttempt && typeof playAttempt.catch === 'function') {
      playAttempt.catch((error) => logDebug('Sound play failed', { url, error }));
    }
  } catch (error) {
    logDebug('Sound init failed', { url, error });
  }
}

function playSoundForAlert() {
  playSoundOnce(state.soundUrl);
}

function normaliseDetail(detail) {
  const event = detail && typeof detail.event === 'object' ? detail.event : {};
  const eventData = event && typeof event.data === 'object' ? event.data : {};

  const mergedData = {
    ...(event || {}),
    ...(eventData || {})
  };

  return {
    listener: detail && detail.listener ? detail.listener : '',
    event,
    data: mergedData
  };
}

function resolveString(payload, keys, fallback = '') {
  const data = payload && payload.data ? payload.data : {};
  const event = payload && payload.event ? payload.event : {};

  for (const key of keys) {
    const value = data[key] ?? event[key];
    if (typeof value === 'string' && value.trim()) return value.trim();
  }
  return fallback;
}

function resolveNumber(payload, keys, fallback = 0) {
  const data = payload && payload.data ? payload.data : {};
  const event = payload && payload.event ? payload.event : {};

  for (const key of keys) {
    const value = data[key] ?? event[key];
    const parsed = Number(value);
    if (Number.isFinite(parsed)) return parsed;
  }
  return fallback;
}

function resolveBoolean(payload, keys, fallback = false) {
  const data = payload && payload.data ? payload.data : {};
  const event = payload && payload.event ? payload.event : {};

  for (const key of keys) {
    const value = data[key] ?? event[key];
    if (typeof value === 'boolean') return value;
    if (value === 'true') return true;
    if (value === 'false') return false;
  }
  return fallback;
}

function resolveUsername(payload, fallback = 'Someone') {
  return resolveString(
    payload,
    ['name', 'username', 'displayName', 'display_name', 'userName', 'senderName'],
    ''
  ) || fallback;
}

function resolveAmountInt(payload, fallback = 1) {
  const n = resolveNumber(payload, ['amount', 'count', 'months', 'viewers'], NaN);
  if (Number.isFinite(n) && n > 0) return Math.floor(n);
  return fallback;
}

function looksPrime(payload) {
  const data = payload && payload.data ? payload.data : {};
  const event = payload && payload.event ? payload.event : {};

  const candidates = [
    data.isPrime,
    data.prime,
    data.plan,
    data.planName,
    data.subPlan,
    data.subscriptionPlan,
    data.tier,
    data.message,
    event.isPrime,
    event.prime,
    event.plan,
    event.planName,
    event.subPlan,
    event.subscriptionPlan,
    event.tier,
    event.message
  ];

  return candidates.some((value) => {
    if (value === true) return true;
    if (typeof value !== 'string' && typeof value !== 'number') return false;
    return String(value).toLowerCase().includes('prime');
  });
}

function isGiftDerivedSubscriberEvent(listener, payload) {
  if (listener === 'subscriber-gifted-latest') return false;

  const data = payload && payload.data ? payload.data : {};
  const event = payload && payload.event ? payload.event : {};

  return Boolean(
    data.sender ||
      data.gifted ||
      data.isGift ||
      data.isGifted ||
      data.bulkGifted ||
      event.sender ||
      event.gifted ||
      event.isGift ||
      event.isGifted ||
      event.bulkGifted
  );
}

function cleanupFingerprints() {
  const now = Date.now();
  for (const [key, timestamp] of state.fingerprints.entries()) {
    if (now - timestamp > 8000) {
      state.fingerprints.delete(key);
    }
  }
}

function buildFingerprint(listener, payload) {
  const data = payload && payload.data ? payload.data : {};
  const event = payload && payload.event ? payload.event : {};

  const parts = [
    listener,
    data._id,
    data.id,
    data.name,
    data.username,
    data.displayName,
    data.sender,
    data.gifted,
    data.amount,
    data.tier,
    data.level,
    data.active,
    data.levelChanged,
    data.message,
    data.createdAt,
    event._id,
    event.id
  ];

  return parts
    .filter((part) => part !== undefined && part !== null && part !== '')
    .join('|');
}

function isDuplicateEvent(listener, payload) {
  cleanupFingerprints();

  const fingerprint = buildFingerprint(listener, payload);
  if (!fingerprint) return false;

  const now = Date.now();
  const previous = state.fingerprints.get(fingerprint);

  state.fingerprints.set(fingerprint, now);

  if (!previous) return false;
  return now - previous < 3000;
}

function buildHypeTrainAlert(payload) {
  const active = resolveBoolean(payload, ['active'], false);
  const level = resolveNumber(payload, ['level'], 0);
  const levelChanged = resolveBoolean(payload, ['levelChanged', 'level_changed'], false);

  const prevActive = state.hypetrainActive;
  const prevLevel = state.hypetrainLevel;

  state.hypetrainActive = active;
  state.hypetrainLevel = level || prevLevel;

  if (active === true && prevActive !== true) {
    return {
      type: 'hypetrain',
      headline: 'Hype Train started'
    };
  }

  if (
    active === true &&
    (levelChanged === true || (Number.isFinite(level) && level > (prevLevel || 0)))
  ) {
    const displayLevel = Number.isFinite(level) && level > 0 ? level : prevLevel || 1;
    return {
      type: 'hypetrain',
      headline: `Hype Train level ${displayLevel}`
    };
  }

  if (active === false && prevActive === true) {
    return {
      type: 'hypetrain',
      headline: 'Hype Train ended'
    };
  }

  return null;
}

function buildAlert(listener, payload) {
  const username = resolveUsername(payload);

  if (listener === 'follower-latest') {
    return {
      type: 'follow',
      headline: `${username} just followed`
    };
  }

  if (listener === 'cheer-latest') {
    const amount = resolveAmountInt(payload, 100);
    return {
      type: 'bits',
      headline: `${username} cheered ${amount} bits`
    };
  }

  if (listener === 'subscriber-gifted-latest') {
    const amount = resolveAmountInt(payload, 1);
    state.giftBulkAt = Date.now();
    state.giftBulkSender = username;
    state.giftBulkCount = amount;
    return null;
  }

  if (
    listener === 'subscriber-latest' ||
    listener === 'subscriber-new-latest' ||
    listener === 'subscriber-resub-latest'
  ) {
    if (isGiftDerivedSubscriberEvent(listener, payload)) {
      const gifter = resolveString(payload, ['sender', 'senderName'], '');
      const recipient = resolveString(payload, ['gifted', 'name'], '');
      const now = Date.now();
      const gifterFallback =
        state.giftBulkAt && now - state.giftBulkAt < 12000 ? state.giftBulkSender : '';

      const finalGifter = gifter || gifterFallback;
      const finalRecipient = recipient || username;

      if (!finalGifter || !finalRecipient) return null;

      return {
        type: 'gifted-subs',
        headline: `${finalRecipient} received a gifted sub from ${finalGifter}`
      };
    }

    if (looksPrime(payload)) {
      return {
        type: 'prime-sub',
        headline: `${username} subscribed with Prime`
      };
    }

    return {
      type: 'subscribed',
      headline: `${username} just subscribed`
    };
  }

  if (listener === 'raid-latest') {
    const amount = resolveAmountInt(payload, 1);
    return {
      type: 'raid',
      headline: `${username} raided with ${amount} viewers`
    };
  }

  if (listener === 'hypetrain-latest') {
    return buildHypeTrainAlert(payload);
  }

  return null;
}

function renderAlert(alert) {
  if (els.alert) els.alert.dataset.type = alert.type;
  if (els.headline) els.headline.textContent = alert.headline;
  if (!els.icon) return;

  const iconUrl = state.iconUrls[alert.type] || state.iconUrls.follow;
  if (!iconUrl) {
    els.icon.style.display = 'none';
    return;
  }
  els.icon.style.display = '';
  els.icon.src = iconUrl;
}

function hideAlert() {
  els.alert.classList.remove('is-active');
}

function resetRowAnimation() {
  if (!els.row) return;
  els.row.style.animation = 'none';
  els.row.style.opacity = '0';
  els.row.style.transform = 'translateY(6px)';
  void els.row.offsetWidth;
  els.row.style.animation = '';
  els.row.style.opacity = '';
  els.row.style.transform = '';
}

function playNext() {
  if (state.active) return;

  const next = state.queue.shift();
  if (!next) return;

  state.active = true;
  renderAlert(next);
  playSoundForAlert();

  hideAlert();
  resetRowAnimation();
  void els.alert.offsetWidth;
  els.alert.classList.add('is-active');

  const fadeAt = Math.max(0, state.duration - 350);

  window.setTimeout(() => {
    hideAlert();
  }, fadeAt);

  window.setTimeout(() => {
    state.active = false;
    playNext();
  }, state.duration);
}

function enqueueAlert(alert) {
  if (!alert) return;
  state.queue.push(alert);
  playNext();
}

function handleIncoming(detail) {
  const payload = normaliseDetail(detail);
  const listener = payload.listener;

  window.TeeweeNotchAlertsLastEvent = detail;
  logDebug('Incoming detail', detail);
  logDebug('Normalised payload', payload);

  const supported = [
    'follower-latest',
    'cheer-latest',
    'subscriber-latest',
    'subscriber-new-latest',
    'subscriber-resub-latest',
    'subscriber-gifted-latest',
    'raid-latest',
    'hypetrain-latest'
  ];

  if (!listener) return;
  if (!supported.includes(listener)) return;
  if (isDuplicateEvent(listener, payload)) {
    logDebug('Skipped duplicate event', { listener, payload });
    return;
  }

  const alert = buildAlert(listener, payload);

  if (!alert) {
    logDebug('Skipped event after buildAlert', { listener, payload });
    return;
  }

  logDebug('Built alert', alert);
  enqueueAlert(alert);
}
